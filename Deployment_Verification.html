Gate 4 — AST Reflection & Method Signature Audit Rule

def test_gate4_ast_method_signature_audit(self):
    """AST Reflection Audit - detect missing 'self'/'cls' in class methods."""
    for fpath in app_files:
        tree = ast.parse(code)
        for node in ast.walk(tree):
            if isinstance(node, ast.ClassDef):
                for item in node.body:
                    if isinstance(item, ast.FunctionDef):
                        is_static = any(d.id == "staticmethod" for d in item.decorator_list)
                        is_class = any(d.id == "classmethod" for d in item.decorator_list)
                        args = [a.arg for a in item.args.args]
                        if item.name == "__new__":
                            assert args[0] in ["cls", "self"]
                        elif not is_static and not is_class:
                            assert args[0] == "self" # Catches missing 'self' at build time!
