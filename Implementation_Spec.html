6. Database Specification (PostgreSQL)

PostgreSQL schema, indexes, connection pools, and retention policies implemented in app/database.py:

Alerts Table Schema

CREATE TABLE IF NOT EXISTS alerts (
    id SERIAL PRIMARY KEY,
    symbol VARCHAR(30) NOT NULL,
    alert_date DATE NOT NULL,
    scanner VARCHAR(50) NOT NULL,
    score NUMERIC(5,2),
    close_price NUMERIC(10,2),
    entry_price NUMERIC(10,2),
    stop_loss NUMERIC(10,2),
    target_price NUMERIC(10,2),
    risk_reward NUMERIC(5,2),
    sector VARCHAR(100),
    metadata JSONB,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    CONSTRAINT unique_alert_per_day UNIQUE (symbol, alert_date, scanner)
);

CREATE INDEX IF NOT EXISTS idx_alerts_date_scanner ON alerts(alert_date, scanner);
CREATE INDEX IF NOT EXISTS idx_alerts_symbol ON alerts(symbol);
