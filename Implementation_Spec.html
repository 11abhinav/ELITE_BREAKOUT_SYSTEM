3. Complete SQL Query Reference

Alert Insertion & Deduplication Query

INSERT INTO alerts (symbol, alert_date, scanner, score, close_price, entry_price, stop_loss, target_price, risk_reward, sector, metadata)
VALUES (%s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s)
ON CONFLICT (symbol, alert_date, scanner) 
DO UPDATE SET 
    score = EXCLUDED.score,
    close_price = EXCLUDED.close_price,
    entry_price = EXCLUDED.entry_price,
    stop_loss = EXCLUDED.stop_loss,
    target_price = EXCLUDED.target_price,
    risk_reward = EXCLUDED.risk_reward,
    metadata = EXCLUDED.metadata,
    created_at = NOW();

Scanner Health Heartbeat Query

INSERT INTO scanner_health (scanner_name, status, last_success, today_alerts, processed_count, total_count, error_msg, updated_at)
VALUES (%s, %s, %s, %s, %s, %s, %s, NOW())
ON CONFLICT (scanner_name)
DO UPDATE SET
    status = EXCLUDED.status,
    last_success = COALESCE(EXCLUDED.last_success, scanner_health.last_success),
    today_alerts = EXCLUDED.today_alerts,
    processed_count = EXCLUDED.processed_count,
    total_count = EXCLUDED.total_count,
    error_msg = EXCLUDED.error_msg,
    updated_at = NOW();
